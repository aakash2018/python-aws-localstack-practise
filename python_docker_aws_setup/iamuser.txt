1. User Create Karo
aws --endpoint-url=http://localhost:4566 \
iam create-user --user-name admin

2. Users List Karo
aws --endpoint-url=http://localhost:4566 \
iam list-users

3. Specific User Details
aws --endpoint-url=http://localhost:4566 \
iam get-user --user-name admin

4. Group Create
aws --endpoint-url=http://localhost:4566 \
iam create-group --group-name Developers

5. User Add to Group
aws --endpoint-url=http://localhost:4566 \
iam add-user-to-group \
--user-name admin \
--group-name Developers

6. Roles List
aws --endpoint-url=http://localhost:4566 \
iam list-roles

7. IAM Groups Dekhne ke Liye
aws --endpoint-url=http://localhost:4566 iam list-groups

8. Role Create Command Run Karo
aws --endpoint-url=http://localhost:4566 iam create-role --role-name EC2Role --assume-role-policy-document file://trust-policy.json

9. verify roles
aws --endpoint-url=http://localhost:4566 iam list-roles

10. EC2 Instance Launch (Real AWS में)
aws ec2 run-instances --image-id ami-12345678 --count 1 --instance-type t2.micro --iam-instance-profile Name=EC2Role
