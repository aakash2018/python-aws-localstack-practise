1. create image using docker
docker build -t fastapi-app:v1 .

//docker build -t fastapi:v1 -f .dockerfile .


2. check image
docker images

3. ECR Repository Create
aws --endpoint-url=http://localhost:4566 ecr create-repository --repository-name fastapi-app

4. List repository
aws --endpoint-url=http://localhost:4566 ecr describe-repositories

5. Repository URI Nikalo
aws --endpoint-url=http://localhost:4566 ecr describe-repositories
